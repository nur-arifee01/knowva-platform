import { GoogleGenAI } from '@google/genai';
import { NextResponse } from 'next/server';

export async function POST(req: Request) {
  const { documentText, prompt } = await req.json();
  // Initialize Google Gemini Client
  const ai = new GoogleGenAI({});
  
  const response = await ai.models.generateContent({
    model: 'gemini-2.5-flash',
    contents: [{ text: `Context: ${documentText}\nTask: ${prompt}` }],
  });

  return NextResponse.json({ result: response.text });
}

ดูย้อนหลังได้ตลอดชีพครับ (Lifetime Access) ไม่มีวันหมดอายุ คุณสามารถจัดสรรเวลาเรียนได้ตามความสะดวกของตนเอง และยังได้รับสิทธิ์เข้าถึงเนื้อหาที่อัปเดตใหม่ในอนาคตฟรี
เวลาติดปัญหาหรือเขียนโค้ดแล้วเกิดบั๊ก มีใครช่วยตอบคำถามไหม?
เรามีห้อง Discord พิเศษสำหรับนักเรียนในคอร์ส โดยจะมีทีมงาน Teaching Assistants (TA) และคุณผู้สอนคอยสแตนด์บายช่วยเหลือ คุณสามารถแปะโค้ดหรือภาพหน้าจอเพื่อรับคำปรึกษาได้ทุกวัน
เมื่อเรียนจบหลักสูตร มีใบประกาศนียบัตร (Certificate) ให้หรือไม่?
มีครับ ผู้เรียนที่ส่งงานและผ่านเกณฑ์การประเมิน Capstone Project จะได้รับ Verified Digital Certificate ที่สามารถนำไปแนบใน LinkedIn หรือยื่นสมัครงานกับบริษัทต่างๆ ได้ทันที
สามารถขอใบกำกับภาษีเต็มรูปสำหรับเบิกบริษัทได้ไหม?
สามารถออกใบกำกับภาษีเต็มรูป (e-Tax Invoice) ได้อย่างถูกต้องตามกฎหมายครับ ในขั้นตอนการชำระเงินสามารถเลือกขอใบเสร็จ/ใบกำกับภาษีในนามบริษัทได้เลยครับ
